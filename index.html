  function fallbackCopy(text, successMsg) {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.style.position = 'fixed';
    ta.style.opacity = '0';
    document.body.appendChild(ta);
    ta.select();
    try {

    } catch (err) {
      showStatus('Nepodarilo sa skopírovať text do schránky.');
    }
    document.body.removeChild(ta);

  function getVoteStats(key) {
    const entry = locationVotes[key] || { up: 0, down: 0, user: null };
    const up = Number(entry.up) || 0;
    const down = Number(entry.down) || 0;

  }

  function toggleVote(key, direction) {
    const cur = locationVotes[key] ? { ...locationVotes[key] } : { up: 0, down: 0, user: null };
    if (cur.user === direction) {
      if (direction === 'up') cur.up = Math.max(0, (cur.up || 0) - 1);
      else cur.down = Math.max(0, (cur.down || 0) - 1);
      cur.user = null;
    } else {
      if (cur.user === 'up') cur.up = Math.max(0, (cur.up || 0) - 1);
      if (cur.user === 'down') cur.down = Math.max(0, (cur.down || 0) - 1);
      if (direction === 'up') cur.up = (cur.up || 0) + 1;
      else cur.down = (cur.down || 0) + 1;
      cur.user = direction;
    }
    locationVotes[key] = cur;
    saveVotes();
  }

  function extractTimeLabel(url) {
    if (!url) return '';
    const match = url.match(/[?&]t=(\d+)/);
    if (!match) return '';
    const totalSec = parseInt(match[1], 10);
    if (isNaN(totalSec) || totalSec === 0) return '0:00';
    const h = Math.floor(totalSec / 3600);
    const m = Math.floor((totalSec % 3600) / 60);
    const s = totalSec % 60;
    if (h > 0) return `${h}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
    return `${m}:${String(s).padStart(2,'0')}`;
  }

  function renderPaginationControls(totalPages) {
    const container = document.getElementById('pagination-controls');
    if (!container) return;
    if (totalPages <= 1) {
      container.innerHTML = '';
      return;
    }

    const parts = [];
    parts.push(`<button type="button" class="btn" data-page="${currentPage - 1}" ${currentPage <= 1 ? 'disabled' : ''}>← Predchádzajúca</button>`);

    const pages = new Set([1, totalPages]);
    for (let p = currentPage - 2; p <= currentPage + 2; p++) {
      if (p >= 1 && p <= totalPages) pages.add(p);
    }
    const sortedPages = Array.from(pages).sort((a, b) => a - b);

    let prev = 0;
    sortedPages.forEach(p => {
      if (prev && p - prev > 1) {
        parts.push(`<span style="padding:0 4px;color:var(--text-muted);">…</span>`);
      }
      parts.push(`<button type="button" class="btn ${p === currentPage ? 'active' : ''}" data-page="${p}">${p}</button>`);
      prev = p;
    });

    parts.push(`<button type="button" class="btn" data-page="${currentPage + 1}" ${currentPage >= totalPages ? 'disabled' : ''}>Ďalšia →</button>`);
    container.innerHTML = parts.join('');
  }

  function applyTileLayer(mode) {
    if (!map) return;
    currentTileMode = mode;
    if (currentTileLayer) {
      try { map.removeLayer(currentTileLayer); } catch (e) {}
    }

    let url = 'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png';
    let attr = '&copy; OpenStreetMap';
    if (mode === 'classic') {
      url = 'https://server.arcgisonline.com/ArcGIS/rest/services/World_Street_Map/MapServer/tile/{z}/{y}/{x}';
      attr = 'Tiles &copy; Esri';
    } else if (mode === 'satellite') {
      url = 'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}';
      attr = 'Tiles &copy; Esri';
    }

    currentTileLayer = L.tileLayer(url, { attribution: attr, maxZoom: 19 }).addTo(map);
  }

    function parseCSVLine(line, sep) {
      const result = [];
      let cur = '';
      let inQuotes = false;
      for (let i = 0; i < line.length; i++) {
        const ch = line[i];
        if (ch === '"') {
          if (inQuotes && line[i + 1] === '"') { cur += '"'; i++; }
          else inQuotes = !inQuotes;
        } else if (ch === sep && !inQuotes) {
          result.push(cur.trim());
          cur = '';
        } else cur += ch;
      }
      result.push(cur.trim());
      return result;
    }

    on('csv-upload', 'change', (e) => {
      const file = e.target.files && e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (ev) => {
        try {
          const text = ev.target.result;
          const lines = text.split(/\r?\n/).filter(l => l.trim().length > 0);
          if (lines.length < 2) {
            showStatus('CSV súbor je prázdny alebo nemá riadky dát.');
            return;
          }
          const sep = (lines[0].match(/;/g) || []).length > (lines[0].match(/,/g) || []).length ? ';' : ',';
          const parsed = [];
          for (let i = 1; i < lines.length; i++) {
            const cols = parseCSVLine(lines[i], sep);
            if (!cols[0]) continue;
            let country = cols[1] || '';
            if (country === 'Thailand') country = 'Thajsko';
            if (country === 'Maldives') country = 'Maldivy';
            const lat = parseFloat((cols[6] || '').replace(',', '.'));
            const lng = parseFloat((cols[7] || '').replace(',', '.'));
            parsed.push({
              n: cols[0] || '', k: country, m: cols[2] || '', c: cols[3] || '',
              s: cols[4] || '', v: cols[5] || '',
              lat: isNaN(lat) ? null : lat, lng: isNaN(lng) ? null : lng
            });
          }
          if (parsed.length > 0) {
            rawData = parsed;
            try { localStorage.setItem('mb_sk_locations_v5', JSON.stringify(parsed)); } catch (err) {}

          }
        } catch (err) {
          showStatus('Nepodarilo sa spracovať CSV súbor.');
        }
      };

    });
